 6# Constants
 7daily_extraction = 1200
 8selling_price = 20000
 9processing_cost_per_ton = 6000
10fixed_cost = 10000000
11royalty_rate = 0.05
12
13# Efficiencies for 7 days
14efficiencies = [0.75, 0.78, 0.8, 0.79, 0.73, 0.74, 0.76]
15
16# Step 1: Calculate daily effective outputs
17daily_outputs = []
18
19for eff in efficiencies:
20    output = daily_extraction * eff
21    daily_outputs.append(output)
22
23# Step 2: Total effective output
24total_effective_output = sum(daily_outputs)
25
26# Step 3: Total extracted (before efficiency loss)
27total_extracted = daily_extraction * 7
28
29# Step 4: Revenue
30total_revenue = total_effective_output * selling_price
31
32# Step 5: Processing cost
33total_processing_cost = total_extracted * processing_cost_per_ton
34
35# Step 6: Royalty
36royalty = royalty_rate * total_revenue
37
38# Step 7: Total cost
39total_cost = total_processing_cost + fixed_cost + royalty
40
41# Step 8: Net profit
42net_profit = total_revenue - total_cost
43
44# OUTPUTS
45print("Daily Effective Outputs:")
46for i, val in enumerate(daily_outputs, start=1):
